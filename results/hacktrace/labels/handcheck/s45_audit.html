    # Find all pairs of coders
    total = 0
    for i in range(1, n + 1):
        for j in range(i + 1, n + 1):
            # Count how many coders mentioned at least one of i or j
            count = 0
            for k in range(n):
                x, y = edges[k]
                if x == i or x == j or y == i or y == j:

def main():
    import sys

    input = sys.stdin.read
    data = input().splitlines()

    n, m = map(int, data[0].split())
    statements = data[1:]

    # For each possible criminal, check if the number of truths matches m
    possible_culprits = []
    for criminal in range(1, n + 1):
        truth_count = 0
        for i in range(n):
            stmt = statements[i]
            if stmt[0] == "+":
                if criminal == int(stmt[1:]):
                    truth_count += 1
            else:
                if criminal != int(stmt[1:]):
                    truth_count += 1
        if truth_count == m:
            possible_culprits.append(criminal)

    # Now determine for each suspect whether they must be telling the truth, lying, or not defined
    result = []
    for i in range(n):
        is_truth = False
        is_lie = False
        for criminal in possible_culprits:
            stmt = statements[i]
            if stmt[0] == "+":
                if criminal == int(stmt[1:]):
                    is_truth = True
                else:
                    is_lie = True
            else:
                if criminal != int(stmt[1:]):
                    is_truth = True
                else:
                    is_lie = True
        if is_truth and not is_lie:
            result.append("Truth")
        elif is_lie and not is_truth:
            result.append("Lie")
        else:
            result.append("Not defined")

    print("\n".join(result))

def count_good_pairs(n, a, b):
    # Calculate the difference for each topic
    diff = [a[i] - b[i] for i in range(n)]

    # Sort the differences
    diff.sort()

    # Use two pointers to count the number of good pairs
    count = 0
    left, right = 0, n - 1

    while left < right:
        if diff[left] + diff[right] > 0:
            count += right - left
            right -= 1
        else:
            left += 1

    return count

    def dfs(s, c):
        if (s, c) in memo:

def main():
    import sys

    n, m, k = map(int, input().split())

    if k == 0:

    for i in range(2, n + 1):

def is_c_good(s, c):
    n = len(s)
    if n == 1:
        return s == c

def min_moves_to_a_good(s):
    n = len(s)

    # Memoization cache
    memo = {}

    def dfs(s, c):
        if s in memo:
            return memo[s]
        n = len(s)
        if n == 1:
            if s == c:
                memo[s] = 0
                return 0

                memo[s] = 1

def main():

    t = sys.stdin.readline().strip()

    # Generate all unique permutations of s

import sys


def is_permutation(arr, m):
    if len(arr) != m:
        return False
    seen = set()
    for num in arr:
        if num < 1 or num > m or num in seen:
            return False
        seen.add(num)
    return True

                window = a[i : i + k]
